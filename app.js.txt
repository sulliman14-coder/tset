async function loadSettingsFromDB() {
  showLoading(true);
  try {
    const response = await fetch(`${API_BASE}/get-settings`);
    const result = await response.json();
    
    if (result.success && result.data) {
      const settingsObj = {};
      result.data.forEach(item => {
        if (item.setting_key && item.setting_value !== null) {
          settingsObj[item.setting_key] = item.setting_value;
        }
      });
      
      if (Object.keys(settingsObj).length > 0) {
        appConfig = { ...defaultConfig, ...settingsObj };
        updateUI();
      }
    }
  } catch (error) {
    console.error('Error loading settings:', error);
    showMessage('فشل تحميل الإعدادات - سيتم استخدام الإعدادات الافتراضية');
  } finally {
    showLoading(false);
  }
}

async function saveSettingsToDB(settings) {
  showLoading(true);
  try {
    const response = await fetch(`${API_BASE}/save-settings`, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json'
      },
      body: JSON.stringify({ settings })
    });
    
    const result = await response.json();
    
    if (result.success) {
      showMessage('✅ تم حفظ الإعدادات بنجاح!');
      return true;
    } else {
      showMessage('❌ فشل حفظ الإعدادات: ' + result.error);
      return false;
    }
  } catch (error) {
    console.error('Error saving settings:', error);
    showMessage('❌ حدث خطأ أثناء الحفظ');
    return false;
  } finally {
    showLoading(false);
  }
}

// ============================================
// دوال واجهة المستخدم
// ============================================

function showLoading(show) {
  const overlay = document.getElementById('loadingOverlay');
  if (overlay) {
    overlay.classList.toggle('show', show);
  }
}

function showMessage(text) {
  const msg = document.getElementById('successMessage');
  if (msg) {
    msg.textContent = text;
    msg.classList.add('show');
    setTimeout(() => {
      msg.classList.remove('show');
    }, 3000);
  }
}

function updateUI() {
  const appName = document.getElementById('app-name');
  const mainTitle = document.getElementById('main-title');
  const subTitle = document.getElementById('sub-title');
  const promoBadge = document.getElementById('promo-badge');
  const logoEmoji = document.getElementById('logoEmoji');
  const logoImage = document.getElementById('logoImage');
  
  if (appName) appName.textContent = appConfig.coffee_shop_name || defaultConfig.coffee_shop_name;
  if (mainTitle) mainTitle.textContent = appConfig.coffee_shop_name || defaultConfig.coffee_shop_name;
  if (subTitle) subTitle.textContent = appConfig.tagline || defaultConfig.tagline;
  if (promoBadge) promoBadge.textContent = '☕ ' + (appConfig.promo_text || defaultConfig.promo_text);
  
  const logoUrl = appConfig.logo_url || defaultConfig.logo_url;
  if (logoUrl && logoImage && logoEmoji) {
    logoImage.src = logoUrl;
    logoImage.style.display = 'block';
    logoEmoji.style.display = 'none';
    logoImage.onerror = function() {
      this.style.display = 'none';
      logoEmoji.style.display = 'block';
    };
  } else if (logoImage && logoEmoji) {
    logoImage.style.display = 'none';
    logoEmoji.style.display = 'block';
  }
  
  renderHeroBanners();
  renderProducts();
}
